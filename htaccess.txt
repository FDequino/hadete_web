# ============================================================
#  hadete.com · configuración de servidor
#  Apache / LiteSpeed (Hostinger lee este archivo tal cual).
#  Va en la raíz, junto a index.html.
# ============================================================

DirectoryIndex index.html

# ── Página de error propia ──────────────────────────────────
ErrorDocument 404 /404.html

# ── Una sola dirección canónica ─────────────────────────────
#  Fuerza HTTPS y la versión sin www. Si preferís con www,
#  comentá el segundo bloque y descomentá el tercero.
<IfModule mod_rewrite.c>
  RewriteEngine On

  # HTTPS siempre
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]

  # sin www  (hadete.com)
  RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
  RewriteRule ^(.*)$ https://%1/$1 [R=301,L]

  # con www  (www.hadete.com) — alternativa
  # RewriteCond %{HTTP_HOST} !^www\. [NC]
  # RewriteCond %{HTTP_HOST} !^localhost [NC]
  # RewriteRule ^(.*)$ https://www.%{HTTP_HOST}/$1 [R=301,L]

  # /es  ->  /es/   (evita un redirect extra y una URL duplicada)
  RewriteRule ^es$ /es/ [R=301,L]
</IfModule>

# ── Compresión ──────────────────────────────────────────────
#  Las imágenes ya vienen comprimidas: comprimirlas de nuevo
#  gasta CPU y no baja el peso.
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/plain text/xml
  AddOutputFilterByType DEFLATE application/javascript application/json
  AddOutputFilterByType DEFLATE application/xml image/svg+xml
</IfModule>
<IfModule mod_brotli.c>
  AddOutputFilterByType BROTLI_COMPRESS text/html text/css text/plain text/xml
  AddOutputFilterByType BROTLI_COMPRESS application/javascript application/json
  AddOutputFilterByType BROTLI_COMPRESS application/xml image/svg+xml
</IfModule>

# ── Caché ───────────────────────────────────────────────────
#  Imágenes y tipografías: un año, porque sus nombres llevan el
#  ancho y cambian si cambia el archivo.
#  CSS y JS: una semana, que es el plazo en que podés esperar a
#  que un visitante recurrente vea un cambio de estilo.
#  HTML: nunca cacheado, para que una corrección de texto se vea
#  en la próxima visita.
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/avif                "access plus 1 year"
  ExpiresByType image/webp                "access plus 1 year"
  ExpiresByType image/jpeg                "access plus 1 year"
  ExpiresByType image/png                 "access plus 1 year"
  ExpiresByType image/x-icon              "access plus 1 year"
  ExpiresByType image/svg+xml             "access plus 1 year"
  ExpiresByType font/woff2                "access plus 1 year"
  ExpiresByType text/css                  "access plus 1 week"
  ExpiresByType application/javascript    "access plus 1 week"
  ExpiresByType text/html                 "access plus 0 seconds"
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.(avif|webp|jpg|jpeg|png|ico|svg|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "\.(css|js)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>

  # ── Seguridad ─────────────────────────────────────────────
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
</IfModule>

# ── Tipos correctos ─────────────────────────────────────────
<IfModule mod_mime.c>
  AddType image/avif   .avif
  AddType image/webp   .webp
  AddType font/woff2   .woff2
  AddType application/manifest+json .webmanifest
</IfModule>

# ── Nada de lo que no se sirve ──────────────────────────────
<FilesMatch "^(build-es\.py|LEEME\.md|\.gitignore)$">
  Require all denied
</FilesMatch>
RedirectMatch 404 ^/assets/_source/
